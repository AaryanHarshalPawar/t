1. Graph Representation (Adjacency Matrix and Adjacency List)
File: graph_representation.c
==================================================

#include <stdio.h>
#include <stdlib.h>

#define MAX 20

struct Node {
    int vertex;
    struct Node *next;
};

struct Node *list[MAX];
int matrix[MAX][MAX];

struct Node *createNode(int v) {
    struct Node *newNode = (struct Node *)malloc(sizeof(struct Node));
    newNode->vertex = v;
    newNode->next = NULL;
    return newNode;
}

void addEdgeList(int u, int v) {
    struct Node *newNode = createNode(v);
    newNode->next = list[u];
    list[u] = newNode;

    newNode = createNode(u);
    newNode->next = list[v];
    list[v] = newNode;
}

void addEdgeMatrix(int u, int v) {
    matrix[u][v] = 1;
    matrix[v][u] = 1;
}

void printMatrix(int n) {
    int i, j;
    printf("\nAdjacency Matrix:\n   ");
    for (i = 0; i < n; i++)
        printf("%d ", i);
    printf("\n");
    for (i = 0; i < n; i++) {
        printf("%d: ", i);
        for (j = 0; j < n; j++)
            printf("%d ", matrix[i][j]);
        printf("\n");
    }
}

void printList(int n) {
    int i;
    struct Node *temp;
    printf("\nAdjacency List:\n");
    for (i = 0; i < n; i++) {
        printf("%d -> ", i);
        temp = list[i];
        while (temp != NULL) {
            printf("%d ", temp->vertex);
            temp = temp->next;
        }
        printf("\n");
    }
}

int main() {
    int n, e, i, u, v;

    printf("Enter number of vertices: ");
    scanf("%d", &n);
    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < n; i++)
        list[i] = NULL;

    for (i = 0; i < e; i++) {
        printf("Enter edge %d (u v): ", i + 1);
        scanf("%d %d", &u, &v);
        addEdgeMatrix(u, v);
        addEdgeList(u, v);
    }

    printMatrix(n);
    printList(n);

    return 0;
}

2. Breadth First Search (BFS)
File: bfs.c
==================================================

#include <stdio.h>

#define MAX 20

int matrix[MAX][MAX];
int visited[MAX];
int level[MAX];
int queue[MAX];
int front = 0, rear = 0;

void enqueue(int x) {
    queue[rear++] = x;
}

int dequeue() {
    return queue[front++];
}

int isEmpty() {
    return front == rear;
}

void bfs(int start, int n) {
    int i, current;

    visited[start] = 1;
    level[start] = 0;
    enqueue(start);

    printf("\nBFS Traversal Order: ");
    while (!isEmpty()) {
        current = dequeue();
        printf("%d ", current);

        for (i = 0; i < n; i++) {
            if (matrix[current][i] == 1 && visited[i] == 0) {
                visited[i] = 1;
                level[i] = level[current] + 1;
                enqueue(i);
            }
        }
    }
    printf("\n");
}

int main() {
    int n, e, i, u, v, start;

    printf("Enter number of vertices: ");
    scanf("%d", &n);
    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < e; i++) {
        printf("Enter edge %d (u v): ", i + 1);
        scanf("%d %d", &u, &v);
        matrix[u][v] = 1;
        matrix[v][u] = 1;
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    bfs(start, n);

    printf("\nAnalysis (level of each vertex from start):\n");
    for (i = 0; i < n; i++) {
        if (visited[i])
            printf("Vertex %d is at level %d\n", i, level[i]);
        else
            printf("Vertex %d is not reachable\n", i);
    }

    return 0;
}

3. Depth First Search (DFS)
File: dfs.c
==================================================

#include <stdio.h>

#define MAX 20

int matrix[MAX][MAX];
int visited[MAX];
int order[MAX];
int count = 0;
int n;

void dfs(int current) {
    int i;

    visited[current] = 1;
    order[count++] = current;
    printf("%d ", current);

    for (i = 0; i < n; i++) {
        if (matrix[current][i] == 1 && visited[i] == 0)
            dfs(i);
    }
}

int main() {
    int e, i, u, v, start;

    printf("Enter number of vertices: ");
    scanf("%d", &n);
    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < e; i++) {
        printf("Enter edge %d (u v): ", i + 1);
        scanf("%d %d", &u, &v);
        matrix[u][v] = 1;
        matrix[v][u] = 1;
    }

    printf("Enter starting vertex: ");
    scanf("%d", &start);

    printf("\nDFS Traversal Order: ");
    dfs(start);
    printf("\n");

    printf("\nAnalysis (order in which vertices were visited):\n");
    for (i = 0; i < count; i++)
        printf("Visit %d: vertex %d\n", i + 1, order[i]);

    if (count < n)
        printf("\nSome vertices are not reachable from vertex %d\n", start);

    return 0;
}

4. Prim's Algorithm (Minimum Spanning Tree)
File: prim.c
==================================================

#include <stdio.h>

#define MAX 20
#define INF 99999

int graph[MAX][MAX];

int findMinVertex(int key[], int inMST[], int n) {
    int i, min = INF, minIndex = -1;

    for (i = 0; i < n; i++) {
        if (inMST[i] == 0 && key[i] < min) {
            min = key[i];
            minIndex = i;
        }
    }
    return minIndex;
}

void prim(int n) {
    int parent[MAX], key[MAX], inMST[MAX];
    int i, count, u, v, total = 0;

    for (i = 0; i < n; i++) {
        key[i] = INF;
        inMST[i] = 0;
    }

    key[0] = 0;
    parent[0] = -1;

    for (count = 0; count < n; count++) {
        u = findMinVertex(key, inMST, n);
        if (u == -1)
            break;
        inMST[u] = 1;

        for (v = 0; v < n; v++) {
            if (graph[u][v] != 0 && inMST[v] == 0 && graph[u][v] < key[v]) {
                key[v] = graph[u][v];
                parent[v] = u;
            }
        }
    }

    printf("\nEdges in the Minimum Spanning Tree:\n");
    printf("Edge\tWeight\n");
    for (i = 1; i < n; i++) {
        if (inMST[i]) {
            printf("%d - %d\t%d\n", parent[i], i, graph[i][parent[i]]);
            total = total + graph[i][parent[i]];
        }
    }
    printf("\nTotal weight of MST = %d\n", total);
}

int main() {
    int n, e, i, u, v, w;

    printf("Enter number of vertices: ");
    scanf("%d", &n);
    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < e; i++) {
        printf("Enter edge %d (u v weight): ", i + 1);
        scanf("%d %d %d", &u, &v, &w);
        graph[u][v] = w;
        graph[v][u] = w;
    }

    prim(n);

    return 0;
}

5. Floyd-Warshall Algorithm (All-Pairs Shortest Paths)
File: floyd_warshall.c
==================================================

#include <stdio.h>

#define MAX 20
#define INF 99999

int dist[MAX][MAX];

void floydWarshall(int n) {
    int i, j, k;

    for (k = 0; k < n; k++) {
        for (i = 0; i < n; i++) {
            for (j = 0; j < n; j++) {
                if (dist[i][k] != INF && dist[k][j] != INF &&
                    dist[i][k] + dist[k][j] < dist[i][j]) {
                    dist[i][j] = dist[i][k] + dist[k][j];
                }
            }
        }
    }
}

void printMatrix(int n) {
    int i, j;

    printf("\nAll-Pairs Shortest Path Matrix:\n    ");
    for (i = 0; i < n; i++)
        printf("%5d", i);
    printf("\n");

    for (i = 0; i < n; i++) {
        printf("%3d:", i);
        for (j = 0; j < n; j++) {
            if (dist[i][j] == INF)
                printf("%5s", "INF");
            else
                printf("%5d", dist[i][j]);
        }
        printf("\n");
    }
}

int main() {
    int n, e, i, j, u, v, w;

    printf("Enter number of vertices: ");
    scanf("%d", &n);
    printf("Enter number of edges: ");
    scanf("%d", &e);

    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            if (i == j)
                dist[i][j] = 0;
            else
                dist[i][j] = INF;
        }
    }

    for (i = 0; i < e; i++) {
        printf("Enter edge %d (from to weight): ", i + 1);
        scanf("%d %d %d", &u, &v, &w);
        dist[u][v] = w;
    }

    floydWarshall(n);
    printMatrix(n);

    return 0;
}

6. Hash Table with Linear Probing
File: hash_linear_probing.c
==================================================

#include <stdio.h>

int main() {
    int table[50];
    int i, n, size, index, key;
    int collisions = 0;

    printf("\nEnter size of the table:  ");
    scanf("%d", &size);

    printf("\nEnter no. of keys: ");
    scanf("%d", &n);

    if(n > size){
        printf("Invalid...");
        return 0;
    }

    for(i = 0; i < size; i++){
        table[i] = -1;
    }

    for(i = 0; i < n; i ++){
        printf("\nEnter key %d: ", i+1);
        scanf("%d", &key);

        index = key % size;

        if(table[index] != -1){
            collisions++;
            printf("\nCollision at %d for key %d.\n", index, key);
            
            while(table[index] != -1){
                index = (index + 1) % size;
            }
            printf("\nKey inserted at %d", index);
        }
        table[index] = key;
    }
    
    printf("\nFinal hash table: \n");
    printf("\tKey\t\tIndex\n");

    for(i = 0; i < size; i++){
        if(table[i] == -1){
            printf("\t-\t\t%d\n", i);
        }
        else{
            printf("\t%d\t\t%d\n", table[i], i);
        }
    }
    printf("\nTotal collisions: %d", collisions);
    return 0;
}

7. Hash Table with Separate Chaining
File: hash_chaining.c
==================================================

#include <stdio.h>
#include <stdlib.h>

struct node {
    int key;
    struct node *next;
};

struct node *table[20];
int size;

void insert(int key) {
    int index = key % size;
    struct node *newnode, *temp;

    newnode = (struct node *)malloc(sizeof(struct node));
    newnode->key = key;
    newnode->next = NULL;

    if (table[index] == NULL) {
        table[index] = newnode;
    } else {
        temp = table[index];
        while (temp->next != NULL)
            temp = temp->next;
        temp->next = newnode;
    }

    printf("Key %d inserted at bucket %d\n", key, index);
}

void search(int key) {
    int index = key % size;
    int pos = 1;
    struct node *temp = table[index];

    while (temp != NULL) {
        if (temp->key == key) {
            printf("Key %d found in bucket %d at position %d\n", key, index, pos);
            return;
        }
        temp = temp->next;
        pos++;
    }
    printf("Key %d not found\n", key);
}

void deleteKey(int key) {
    int index = key % size;
    struct node *temp = table[index];
    struct node *prev = NULL;

    while (temp != NULL) {
        if (temp->key == key) {
            if (prev == NULL)
                table[index] = temp->next;
            else
                prev->next = temp->next;
            free(temp);
            printf("Key %d deleted from bucket %d\n", key, index);
            return;
        }
        prev = temp;
        temp = temp->next;
    }
    printf("Key %d not found\n", key);
}

void display() {
    int i;
    struct node *temp;

    printf("\nHash Table:\n");
    for (i = 0; i < size; i++) {
        printf("Bucket %d: ", i);
        temp = table[i];
        while (temp != NULL) {
            printf("%d -> ", temp->key);
            temp = temp->next;
        }
        printf("NULL\n");
    }
}

int main() {
    int i, choice, key;

    printf("Enter the size of hash table: ");
    scanf("%d", &size);

    for (i = 0; i < size; i++)
        table[i] = NULL;

    while (1) {
        printf("\n1.Insert 2.Search 3.Delete 4.Display 5.Exit\n");
        printf("Enter your choice: ");
        scanf("%d", &choice);

        if (choice == 1) {
            printf("Enter key: ");
            scanf("%d", &key);
            insert(key);
        } else if (choice == 2) {
            printf("Enter key to search: ");
            scanf("%d", &key);
            search(key);
        } else if (choice == 3) {
            printf("Enter key to delete: ");
            scanf("%d", &key);
            deleteKey(key);
        } else if (choice == 4) {
            display();
        } else if (choice == 5) {
            break;
        } else {
            printf("Invalid choice\n");
        }
    }

    return 0;
}
